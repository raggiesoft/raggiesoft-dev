msi.* & prereq.* — Direct 64-Bit MSI SQL Engine, .MST Transforms & Runtime Presets

RaggieScript gives you both worlds: you can write pure zero-MSI native installers (where Error 1618 mutex contention is impossible), or you can drive and author Windows Installer .msi, .msm, .mst, and .msp relational SQL databases directly in native 64-bit (x64 / ARM64) via msi.*!

use installer;

let db = fs.tempDir("msi-demo") / "Sample.msi";
msi.createDatabase(db);
msi.setSummary(db, {
    title:    "Installation Database",
    template: sys.isArm64() ? "Arm64;1033" : "x64;1033",
    schema:   500
});
msi.setProperty(db, "Manufacturer", "RaggieSoft");
msi.execSql(db, [
    "CREATE TABLE `CustomPolicy` (`Key` CHAR(64) NOT NULL, `Value` CHAR(128) NOT NULL PRIMARY KEY `Key`)",
    "INSERT INTO `CustomPolicy` (`Key`, `Value`) VALUES ('Telemetry', 'Disabled')"
]);
const rows = msi.query(db, "SELECT `Key`, `Value` FROM `CustomPolicy`");
assert(rows[0].Value == "Disabled");

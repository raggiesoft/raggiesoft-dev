ui.*, wizard.* & showcase.* — Declarative Win32 Wizard UI, TaskDialogs & Canvas

RaggieScript v1.0 ships two complementary native Win32 UI layers—both Per-Monitor V2 DPI aware, Dark/Light Mode adaptive, Windows Contrast Theme (SPI_GETHIGHCONTRAST) compliant, and 100% UI Automation screen-reader accessible:

  1. 14 Declarative In-Window Wizard Controls (ui.welcome .. ui.readySummary) rendered inside a single continuous wizard window (RsscWizardUiEngine.hpp) across all 11 WizardStyle themes.
  2. Modal Windows 11 TaskDialogIndirect & Modern IFileOpenDialog Prompts (ui.taskDialog, ui.pickFolder, etc.) for one-step micro-installers and standalone utilities.

param enableProxy: bool   = false;
param proxyPort:   string = "8080";

const checkPort = (raw: string) => {
    const p = int(str.trim(raw));
    return (p >= 1024 && p <= 65535)
        ? Validation.ok()
        : Validation.error("Enter a TCP port between 1024 and 65535.");
};

wizard {
    step Network {
        ui.checkbox(text: "Route updates through a local &proxy", bind: ref enableProxy);
        ui.textBox(
            label:    "Local proxy &port:",
            bind:     ref proxyPort,
            when:     () => enableProxy,
            validate: checkPort
        );
    }
}
